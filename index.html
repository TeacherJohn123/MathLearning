const SHEET_NAME = 'Data'; /** * ================================ * GET：讀取雲端資料 * ================================ */ function doGet(e) { try { const sheet = getOrCreateSheet(); const data = sheet.getRange('A1').getValue(); let parsedData = {}; if (data) { try { parsedData = JSON.parse(data); } catch (err) { parsedData = {}; } } return createResponse({ status: 'success', data: parsedData }); } catch (err) { return createResponse({ status: 'error', message: '讀取資料失敗：' + err.message }); } } /** * ================================ * POST：處理所有前端請求 * ================================ */ function doPost(e) { if (!e || !e.postData || !e.postData.contents) { return createResponse({ status: 'error', message: '未接收到 POST 資料' }); } let req; try { req = JSON.parse(e.postData.contents); } catch (err) { return createResponse({ status: 'error', message: 'JSON 格式解析失敗：' + err.message }); } const action = req.action; /** * ================================ * 1. 管理員登入 * ================================ */ if (action === 'verifyAdmin') { const correctPwd = PropertiesService .getScriptProperties() .getProperty('ADMIN_PASSWORD'); if (!correctPwd) { return createResponse({ status: 'error', message: '後端尚未設定 ADMIN_PASSWORD' }); } if (String(req.password) === String(correctPwd)) { return createResponse({ status: 'success', token: 'math_admin_v3_token' }); } return createResponse({ status: 'error', message: '密碼錯誤' }); } /** * ================================ * 2. 儲存資料 * ================================ */ if (action === 'saveData') { if (req.token !== 'math_admin_v3_token') { return createResponse({ status: 'error', message: '未經授權的寫入請求' }); } if (!req.data || typeof req.data !== 'object') { return createResponse({ status: 'error', message: '沒有收到有效的資料' }); } try { const sheet = getOrCreateSheet(); const jsonData = JSON.stringify(req.data); sheet.getRange('A1').setValue(jsonData); SpreadsheetApp.flush(); return createResponse({ status: 'success', message: '資料已成功儲存至 Google Sheets', savedAt: new Date().toISOString() }); } catch (err) { return createResponse({ status: 'error', message: 'Google Sheets 儲存失敗：' + err.message }); } } /** * ================================ * 3. AI 議案助手 * ================================ */ if (action === 'askAI') { const apiKey = PropertiesService .getScriptProperties() .getProperty('GEMINI_API_KEY'); if (!apiKey) { return createResponse({ status: 'error', message: '後端未設定 GEMINI_API_KEY' }); } const url = 'https://generativelanguage.googleapis.com/' + 'v1beta/models/gemini-1.5-flash:generateContent?key=' + encodeURIComponent(apiKey); const payload = { contents: [ { role: 'user', parts: [ { text: String(req.prompt || '') } ] } ], generationConfig: { temperature: 0.4 } }; const options = { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true }; try { const response = UrlFetchApp.fetch(url, options); const json = JSON.parse(response.getContentText()); if (json.error) { return createResponse({ status: 'error', message: json.error.message }); } if ( !json.candidates || !json.candidates[0] || !json.candidates[0].content || !json.candidates[0].content.parts ) { return createResponse({ status: 'error', message: 'AI 沒有回傳有效內容' }); } const text = json.candidates[0] .content .parts .map(p => p.text || '') .join(''); return createResponse({ status: 'success', data: text }); } catch (err) { return createResponse({ status: 'error', message: 'AI 服務連線失敗：' + err.message }); } } /** * ================================ * 未知 action * ================================ */ return createResponse({ status: 'error', message: '未知的請求動作：' + action }); } /** * ================================ * 取得或建立工作表 * ================================ */ function getOrCreateSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } return sheet; } /** * ================================ * JSON Response * ================================ */ function createResponse(obj) { return ContentService .createTextOutput( JSON.stringify(obj) ) .setMimeType( ContentService.MimeType.JSON ); }
